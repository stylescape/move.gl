Usage

// Filter Mixins .image-blur { @include filter_blur(4px); } .image-grayscale { @include filter_grayscale(100%); } .image-brightness { @include filter_brightness(150%); } // Shadow Mixins .card { @include elevation(3); } .box { @include box_shadow(0, 4px, 8px, 0, rgba(0,0,0, .2)); } // Opacity Mixin .faded { @include set_opacity(0.5); } // Opacity Transition .image { @include opacity_transition(0.3s, 0.75); }
